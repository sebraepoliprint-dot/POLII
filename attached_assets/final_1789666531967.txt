const EstoqueCliente = ({ aoVoltar, salvarPedido, catalogo, atualizarCatalogo }) => {
const [carrinho, setCarrinho] = useState({}); const [setorCliente, setSetorCliente] = useState('');
const totalPecas = Object.values(carrinho).reduce((acc, item) => acc + item.qtd, 0);

const enviarPedido = () => {
if (totalPecas === 0) return alert("Adicione pelo menos 1 item ao pedido.");
if (!setorCliente) return alert("Por favor, informe qual é o seu setor.");
let listaItens = [];
for (const catId in carrinho) {
const item = carrinho[catId];
if (item.qtd > 0 && item.modeloId) {
const categoria = catalogo.find(c => c.id === catId); const modeloObj = categoria.modelos.find(m => m.id === item.modeloId);
listaItens.push({ nome: categoria.nome, qtd: item.qtd, modelo: modeloObj.nome });
}
}
if (listaItens.length === 0) return alert("Selecione os modelos das peças que deseja pedir.");
const novoCatalogo = catalogo.map(cat => {
const itemCarrinho = carrinho[cat.id];
if (itemCarrinho && itemCarrinho.qtd > 0 && itemCarrinho.modeloId) { return { ...cat, modelos: cat.modelos.map(m => m.id === itemCarrinho.modeloId ? { ...m, estoque: m.estoque - itemCarrinho.qtd } : m) }; }
return cat;
});
atualizarCatalogo(novoCatalogo);
salvarPedido({ id: Math.floor(Math.random() * 9000), setor: setorCliente, itens: listaItens, status: 'Pendente' });
alert("✅ Pedido enviado! O estoque já foi atualizado."); aoVoltar();
};

const etapasPonto = ['Entrada', 'Saída Almoço', 'Volta Almoço', 'Fim Expediente'];
const proximaEtapaIndex = historicoPontosTecnico.filter(i => i.tipo !== 'Saída Informada' && i.tipo !== 'Volta (Saída Informada)').length;
const etapaAtual = etapasPonto[proximaEtapaIndex] || 'Jornada Concluída';

const executarRegistro = (tipoNome) => {
const salvar = (lat, lng) => { registrarPontoGlobal({ id: Math.random(), tecnico: usuarioLogado, tipo: tipoNome, horario: new Date().toLocaleTimeString('pt-BR'), data: new Date().toLocaleDateString('pt-BR'), coords: `${lat.toFixed(4)}, ${lng.toFixed(4)}`, foto: 'https://images.unsplash.com/photo-1507003211169-0a1dd7228f2d?w=150&q=80' }); alert(`✅ ${tipoNome} registrada com sucesso!`); };
if (navigator.geolocation) { navigator.geolocation.getCurrentPosition((pos) => salvar(pos.coords.latitude, pos.coords.longitude), () => salvar(-3.1190, -60.0217), { enableHighAccuracy: false, timeout: 10000 }); } else salvar(-3.1190, -60.0217);
};

return (
<SafeAreaView style={styles.containerApp}>
<View style={[styles.headerSimples, styles.efeito3D]}><TouchableOpacity onPress={aoVoltar}><Text style={styles.textoVoltar}>⬅ Voltar</Text></TouchableOpacity><Text style={styles.tituloHeader}>Ponto Eletrônico</Text><View style={{width: 50}} /></View>
<ScrollView>
<View style={[styles.cardSessao, styles.efeito3D]}><Text style={styles.textoDataHora}>📅 Hoje: {new Date().toLocaleDateString('pt-BR')}</Text><Text style={styles.subInfo}>{proximaEtapaIndex < 4 ? `Próxima: ${etapaAtual}` : "Jornada Concluída!"}</Text></View>
{proximaEtapaIndex < 4 && <TouchableOpacity style={styles.sombraBotao} onPress={() => { if (proximaEtapaIndex >= 4) return alert("Concluído!"); executarRegistro(etapaAtual); }}><View style={styles.botaoVerdePonto}><Text style={styles.textoBotaoBranco}>Registrar {etapaAtual}</Text></View></TouchableOpacity>}
<TouchableOpacity style={styles.sombraBotao} onPress={() => { const tipo = emSaidaInformada ? 'Volta (Saída Informada)' : 'Saída Informada'; executarRegistro(tipo); setEmSaidaInformada(!emSaidaInformada); }}><View style={[styles.botaoLaranjaPonto, emSaidaInformada && {backgroundColor: '#17a2b8'}]}><Text style={styles.textoBotaoBranco}>{emSaidaInformada ? "🔄 Registrar Volta" : "⚠️ Registrar Saída Informada"}</Text></View></TouchableOpacity>
<Text style={styles.tituloSecao}>📋 Registros de Hoje:</Text>
{historicoPontosTecnico.map((item) => (<View key={item.id} style={[styles.cartaoChamadoLista, styles.efeito3D, {flexDirection: 'row', alignItems: 'center'}]}><View style={{flex: 1}}><Text style={styles.textoIDLista}>{item.tipo}</Text><Text style={{fontSize: 12, color: '#555'}}>GPS: {item.coords}</Text></View><Text style={{fontSize: 15, fontWeight: '900'}}>{item.horario}</Text></View>))}
</ScrollView>
</SafeAreaView>
);
};

const [bancoDeChamados, setBancoDeChamados] = useState([]);
const [bancoDePedidos, setBancoDePedidos] = useState([]);
const [bancoDePontos, setBancoDePontos] = useState([]);

const [catalogoDePecas, setCatalogoDePecas] = useState([
{ id: '1', icone: '🔲', nome: 'Toner / Cartucho', modelos: [{ id: 'm1', nome: 'Brother TN-1060', estoque: 15 }, { id: 'm2', nome: 'HP 85A', estoque: 3 }] },

]);

const [itemSelecionado, setItemSelecionado] = useState(null);
const [filtroChamado, setFiltroChamado] = useState('Todos');

const handleLogin = (tipo, nome) => { setNomeUsuario(nome); setPerfilReal(tipo); setTelaAtual(`menu_${tipo}`); };

const renderTelas = () => {
if (telaAtual === 'login') return <TelaLogin aoLogar={handleLogin} />;
if (telaAtual === 'menu_cliente') return <MenuCliente nome={nomeUsuario} aoSair={() => setTelaAtual('login')} navegarPara={setTelaAtual} ehGod={perfilReal === 'god'} aoVoltarGod={() => setTelaAtual('menu_god')} />;

'menu_god' : `menu_${perfilReal}`)} registrarPontoGlobal={(p) => setBancoDePontos([p, ...bancoDePontos])} historicoPontosTecnico={bancoDePontos.filter(p => p.tecnico === nomeUsuario)} />;

if (telaAtual === 'lista_chamados_tecnico') return <ListaChamadosTecnico chamados={bancoDeChamados} aoVoltar={() => setTelaAtual('menu_tecnico')} abrirDetalhe={(c) => { setItemSelecionado(c); setTelaAtual('atender_chamado_tecnico'); }} filtro={filtroChamado} />;
if (telaAtual === 'atender_chamado_tecnico') return <AtenderChamadoTecnico chamado={itemSelecionado} aoVoltar={() => setTelaAtual('menu_tecnico')} atualizarChamado={(e) => setBancoDeChamados(bancoDeChamados.map(c => c.id === e.id ? e : c))} usuarioLogado={nomeUsuario} ehGod={perfilReal === 'god'} />;
if (telaAtual === 'menu_estoque') return <MenuEstoque nome={nomeUsuario} aoSair={() => setTelaAtual('login')} navegarPara={setTelaAtual} pedidos={bancoDePedidos} ehGod={perfilReal === 'god'} aoVoltarGod={() => setTelaAtual('menu_god')} />;
if (telaAtual === 'lista_pedidos_estoque') return <ListaPedidosEstoque pedidos={bancoDePedidos} aoVoltar={() => setTelaAtual('menu_estoque')} abrirDetalhe={(p) => { setItemSelecionado(p); setTelaAtual('detalhe_pedido_estoque'); }} />;

if (telaAtual === 'gerenciar_catalogo_estoque') return <GerenciarCatalogoEstoque aoVoltar={() => setTelaAtual('menu_estoque')} catalogo={catalogoDePecas} setCatalogo={setCatalogoDePecas} />;
if (telaAtual === 'menu_god') return <MenuGod nome={nomeUsuario} aoSair={() => setTelaAtual('login')} navegarPara={setTelaAtual} chamados={bancoDeChamados} pedidos={bancoDePedidos} registrosPonto={bancoDePontos} />;
if (telaAtual === 'relatorio_god') return <RelatorioGod aoVoltar={() => setTelaAtual('menu_god')} chamados={bancoDeChamados} pedidos={bancoDePedidos} navegarPara={setTelaAtual} />;
if (telaAtual === 'relatorio_ponto_god') return <RelatorioPontoGod aoVoltar={() => setTelaAtual('menu_god')} registrosPonto={bancoDePontos} abrirDetalheTecnico={(nomeTec) => { setItemSelecionado(nomeTec); setTelaAtual('detalhe_ponto_func'); }} />;

if (telaAtual === 'lista_chamados_god') return <ListaChamadosTecnico chamados={bancoDeChamados} aoVoltar={() => setTelaAtual('relatorio_god')} abrirDetalhe={(c) => { setItemSelecionado(c); setTelaAtual('atender_chamado_god'); }} filtro="Todos" />;
if (telaAtual === 'atender_chamado_god') return <AtenderChamadoTecnico chamado={itemSelecionado} aoVoltar={() => setTelaAtual('lista_chamados_god')} atualizarChamado={(e) => setBancoDeChamados(bancoDeChamados.map(c => c.id === e.id ? e : c))} usuarioLogado={nomeUsuario} ehGod={true} />;
if (telaAtual === 'lista_pedidos_god') return <ListaPedidosEstoque pedidos={bancoDePedidos} aoVoltar={() => setTelaAtual('relatorio_god')} abrirDetalhe={(p) => { setItemSelecionado(p); setTelaAtual('detalhe_pedido_god'); }} />;
if (telaAtual === 'detalhe_pedido_god') return <DetalhePedidoEstoque pedido={itemSelecionado} aoVoltar={() => setTelaAtual('lista_pedidos_god')} atualizarPedido={(e) => setBancoDePedidos(bancoDePedidos.map(p => p.id === e.id ? e : p))} abrirImpressao={(p) => { setItemSelecionado(p); setTelaAtual('comprovante_oficial_god'); }} />;
if (telaAtual === 'comprovante_oficial_god') return <ComprovanteEntregaOficial pedido={itemSelecionado} aoVoltar={() => setTelaAtual('detalhe_pedido_god')} />;
};

return <View style={{flex: 1}}>{renderTelas()}</View>;
}
